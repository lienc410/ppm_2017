        public void setMonthlyDates()
        {
            int startMonth = startDate.Month;
            int startYear = startDate.Year;

            int endMonth = asOfDate.Month;
            int endYear = asOfDate.Year;
            if (endMonth > 1)
                endMonth--;
            else if (endMonth == 1)
            {
                endMonth = 12;
                endYear--;
            }

            DateTime startMonthDate = new DateTime(startYear, startMonth, 1);
            DateTime endMonthDate = new DateTime(endYear, endMonth, 1);

            monthlyDates = new List<DateTime>();
            monthlyDatesForMortgageRate30Yr = new List<DateTime>();
            numMonths = (endYear - startYear) * 12 + endMonth - startMonth + 1;
            numMonthsForMortgageRate30Yr = (endYear - startYear + moreYrsForMortgageRate30Yr) * 12 + endMonth - startMonth + 1;

            if (numMonths < 0)
                Console.WriteLine("Please enter the correct date!");

            DateTime intermediateMonthDate = startMonthDate;

            for (int i = 0; i < numMonths; i++)
            {
                //Console.WriteLine(intermediateMonthDate);
                monthlyDates.Add(intermediateMonthDate);
                int intermediateMonth = intermediateMonthDate.Month;
                int intermediateYear = intermediateMonthDate.Year;

                if (intermediateMonth <= 11)
                    intermediateMonth++;
                else if (intermediateMonth == 12)
                {
                    intermediateMonth = 1;
                    intermediateYear++;
                }

                intermediateMonthDate = new DateTime(intermediateYear, intermediateMonth, 1);
            }

            intermediateMonthDate = startMonthDate.AddYears(-moreYrsForMortgageRate30Yr);

            for (int i = 0; i < numMonthsForMortgageRate30Yr; i++)
            {
                //Console.WriteLine(intermediateMonthDate);
                monthlyDatesForMortgageRate30Yr.Add(intermediateMonthDate);
                int intermediateMonth = intermediateMonthDate.Month;
                int intermediateYear = intermediateMonthDate.Year;

                if (intermediateMonth <= 11)
                    intermediateMonth++;
                else if (intermediateMonth == 12)
                {
                    intermediateMonth = 1;
                    intermediateYear++;
                }

                intermediateMonthDate = new DateTime(intermediateYear, intermediateMonth, 1);
            }
        }

        public void calcHistoricalRates(ref List<TimeSeries> histRates, int dataLength)
        {

	        List<DateTime> mDates;
            List<double> avgRates = new List<double>();

            if (histRates != mortgageRates30Yr)
                mDates = monthlyDates;
            else
                mDates = monthlyDatesForMortgageRate30Yr;

            for (int i = 0; i < dataLength; i++)
            {
                double tmpSum = 0;
                int tmpCount = 0;
                foreach (TimeSeries t in histRates)
                {
                    if (i < dataLength - 1)
                        {
                        if (DateTime.Compare(t.AsOfDate, mDates[i]) >= 0 && DateTime.Compare(t.AsOfDate, mDates[i + 1]) < 0)
                            {
                                tmpSum = tmpSum + t.SeriesNumValue;
                                tmpCount++;
                            }
                        }
                    else if (DateTime.Compare(t.AsOfDate, mDates[i]) >= 0)
                    {
                        tmpSum = tmpSum + t.SeriesNumValue;
                        tmpCount++;
                    }


                }

                
//                if (tmpCount > 0)
                    avgRates.Add(tmpSum / tmpCount);
            }

            // for (int i = 0; i < numMonths; i++)
            //Console.WriteLine(avgRates[i]);

            histRates.Clear();


           for (int i = 0; i < dataLength; i++)
               histRates.Add(new TimeSeries("", "", "", "", "", "", mDates[i], mDates[i], avgRates[i], mDates[i]));
//                histRates.Add(new TimeSeries("", "", "", "", "", mDates[i], mDates[i], mDates[i]));

        }

        private List<TimeSeries> getTimeSeriesData(string sqlCmd)
        {
            //log.Info("getTimeSeriesData(string sqlCmd<" + sqlCmd + ">");

            //SACommand cmd = conIQ.CreateCommand();
            SqlCommand cmd = conMs.CreateCommand();
            cmd.CommandText = sqlCmd;
            List<TimeSeries> rs = new List<TimeSeries>();
            try
            {
                //SADataReader dr = cmd.ExecuteReader();
                SqlDataReader dr = cmd.ExecuteReader();
                while (dr.Read())
                {
                    rs.Add(new TimeSeries(
                        "", "",
                        "",
                        ""
                        , "", "", dr.GetDateTime(0), dr.GetDateTime(0), dr.GetDouble(1), dr.GetDateTime(0)));
                }
                dr.Close();
            }
            catch (SAException ex)
            {
                log.Info(ex.Message);
            }
            return rs;
        }
